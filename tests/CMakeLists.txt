cmake_minimum_required(VERSION 3.20)
project(grok_sdk_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CJSON_DIR "$ENV{IDF_PATH}/components/json/cJSON" CACHE PATH "cJSON source directory")
if(NOT EXISTS "${CJSON_DIR}/cJSON.c")
  message(FATAL_ERROR "Set IDF_PATH or CJSON_DIR to the cJSON source directory")
endif()
add_library(cjson STATIC "${CJSON_DIR}/cJSON.c")
target_include_directories(cjson PUBLIC "${CJSON_DIR}")
add_library(grok_bot STATIC ../src/grok_bot.cpp ../src/avatar.cpp)
target_include_directories(grok_bot PUBLIC ../include)
target_link_libraries(grok_bot PRIVATE cjson)
target_compile_options(grok_bot PRIVATE -Wall -Wextra -Werror)
add_executable(sdk_tests sdk_test.cpp)
target_link_libraries(sdk_tests PRIVATE grok_bot cjson)
enable_testing()
add_test(NAME sdk COMMAND sdk_tests)
